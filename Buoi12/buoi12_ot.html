1 Luồng MVC - Cập nhật dữ liệu

Update theo mô hình MVC:

  • View: Hiển thị form cập nhật (đã có dữ liệu cũ)
  • Controller: Nhận ID → Lấy dữ liệu → Hiển thị form → Xử lý POST cập nhật
  • Model: Thêm phương thức update() và findById()

3 Cập nhật User

1. Thêm method vào Model - app/Models/User.php 

// Lấy user theo ID
public function findById($id) {
    $sql = "SELECT * FROM users WHERE id = ?";
    $stmt = $this->conn->prepare($sql);
    $stmt->execute([$id]);
    return $stmt->fetch(PDO::FETCH_ASSOC);
}

// Cập nhật user
public function update($id, $name, $email) {
    $sql = "UPDATE users 
            SET name = ?, email = ?, updated_at = NOW() 
            WHERE id = ?";
    
    $stmt = $this->conn->prepare($sql);
    return $stmt->execute([$name, $email, $id]);
}

2. Controller - app/Controllers/UserController.php 

class UserController {

    private $userModel;

    public function __construct() {
        $this->userModel = new User();
    }

    // ... các method cũ ...

    // Hiển thị form sửa
    public function edit() {
        $id = $_GET['id'] ?? null;
        if (!$id) {
            echo "ID không hợp lệ!";
            return;
        }

        $user = $this->userModel->findById($id);
        if (!$user) {
            echo "Không tìm thấy user!";
            return;
        }

        require_once __DIR__ . '/../Views/user/sua.php';
    }

    // Xử lý cập nhật
    public function update() {
        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $id    = $_POST['id'] ?? null;
            $name  = trim($_POST['name']);
            $email = trim($_POST['email']);

            if ($id) {
                $result = $this->userModel->update($id, $name, $email);
                if ($result) {
                    header("Location: /");
                    exit;
                } else {
                    echo "Lỗi khi cập nhật dữ liệu!";
                }
            }
        }
    }
}
