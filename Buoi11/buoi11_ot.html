2 Hiển thị danh sách Users

1. Model - app/Models/User.php 

class User {
    private $conn;

    public function __construct() {
        $db = new Database();
        $this->conn = $db->getConnection();
    }

    public function getAll() {
        $sql = "SELECT * FROM users ORDER BY id DESC";
        $stmt = $this->conn->prepare($sql);
        $stmt->execute();
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }
}

2. Controller - app/Controllers/UserController.php 

class UserController {

    private $userModel;

    public function __construct() {
        $this->userModel = new User();
    }

    // Hiển thị danh sách user
    public function index() {
        $users = $this->userModel->getAll();
        
        // Truyền dữ liệu sang View
        require_once __DIR__ . '/../Views/user/danhsach.php';
    }
}

2. Controller - app/Controllers/UserController.php 

class UserController {

    private $userModel;

    public function __construct() {
        $this->userModel = new User();
    }

    // Hiển thị form thêm mới
    public function create() {
        require_once __DIR__ . '/../Views/user/themmoi.php';
    }

    // Xử lý thêm mới
    public function store() {
        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $name  = trim($_POST['name']);
            $email = trim($_POST['email']);

            $result = $this->userModel->create($name, $email);

            if ($result) {
                header("Location: /");
                exit;
            } else {
                echo "Lỗi khi thêm dữ liệu!";
            }
        }
    }
}

3. Method create() trong Model - app/Models/User.php 

public function create($name, $email) {
    $sql = "INSERT INTO users (name, email, created_at) 
            VALUES (?, ?, NOW())";
    
    $stmt = $this->conn->prepare($sql);
    return $stmt->execute([$name, $email]);
}
