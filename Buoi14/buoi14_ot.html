2. Controller - app/Controllers/UserController.php 

// Xử lý thêm mới có upload ảnh
public function store() {
    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        $name  = trim($_POST['name']);
        $email = trim($_POST['email']);
        $imagePath = null;

        // Xử lý upload ảnh
        if (isset($_FILES['image']) && $_FILES['image']['error'] === 0) {
            $imagePath = $this->uploadImage($_FILES['image']);
        }

        $result = $this->userModel->createWithImage($name, $email, $imagePath);

        if ($result) {
            header("Location: /");
            exit;
        } else {
            echo "Lỗi khi thêm dữ liệu!";
        }
    }
}

// Hàm upload ảnh (thêm vào Controller)
private function uploadImage($file) {
    $targetDir = __DIR__ . "/../../public/images/";
    if (!is_dir($targetDir)) mkdir($targetDir, 0755, true);

    $fileExt = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    $allowed = ['jpg', 'jpeg', 'png', 'gif'];

    if (!in_array($fileExt, $allowed)) {
        echo "Chỉ chấp nhận file ảnh!";
        return null;
    }

    $newFileName = uniqid() . '.' . $fileExt;
    $targetFile = $targetDir . $newFileName;

    if (move_uploaded_file($file['tmp_name'], $targetFile)) {
        return 'images/' . $newFileName; // Đường dẫn tương đối
    }
    return null;
}

4 Cập nhật User & Ảnh

Model - update method

public function updateWithImage($id, $name, $email, $imagePath = null) {
    if ($imagePath !== null) {
        $sql = "UPDATE users SET name = ?, email = ?, image = ?, updated_at = NOW() WHERE id = ?";
        $stmt = $this->conn->prepare($sql);
        return $stmt->execute([$name, $email, $imagePath, $id]);
    } else {
        $sql = "UPDATE users SET name = ?, email = ?, updated_at = NOW() WHERE id = ?";
        $stmt = $this->conn->prepare($sql);
        return $stmt->execute([$name, $email, $id]);
    }
}

Controller - update method 

public function update() {
    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        $id    = $_POST['id'] ?? null;
        $name  = trim($_POST['name']);
        $email = trim($_POST['email']);
        $oldImage = $_POST['old_image'] ?? null; // hidden input

        $newImagePath = null;
        if (isset($_FILES['image']) && $_FILES['image']['error'] === 0) {
            $newImagePath = $this->uploadImage($_FILES['image']);
            
            // Xoá ảnh cũ nếu có ảnh mới
            if ($newImagePath && $oldImage) {
                $this->deleteImageFile($oldImage);
            }
        }

        $result = $this->userModel->updateWithImage($id, $name, $email, $newImagePath);

        if ($result) {
            header("Location: /");
            exit;
        }
    }
}

// Hàm xoá file ảnh
private function deleteImageFile($imagePath) {
    if ($imagePath) {
        $fullPath = __DIR__ . "/../../public/" . $imagePath;
        if (file_exists($fullPath)) {
            unlink($fullPath);
        }
    }
}

5 Xoá User & Xoá ảnh

Model - delete method (Cập nhật) 

public function delete($id) {
    // Lấy thông tin ảnh trước khi xoá
    $user = $this->findById($id);
    if ($user && !empty($user['image'])) {
        // Xoá file ảnh
        $fullPath = __DIR__ . "/../../public/" . $user['image'];
        if (file_exists($fullPath)) {
            unlink($fullPath);
        }
    }

    $sql = "DELETE FROM users WHERE id = :id";
    $stmt = $this->conn->prepare($sql);
    $stmt->bindParam(':id', $id, PDO::PARAM_INT);
    return $stmt->execute();
}
