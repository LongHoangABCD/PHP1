3 Xoá dữ liệu

1. Model - app/Models/User.php (Thêm method delete) 

public function delete($id) {
    $sql = "DELETE FROM users WHERE id = :id";
    
    $stmt = $this->conn->prepare($sql);
    
    $stmt->bindParam(':id', $id, PDO::PARAM_INT);
    
    return $stmt->execute();
}

2. Controller - app/Controllers/UserController.php (Thêm method delete) 

class UserController {

    private $userModel;

    public function __construct() {
        $this->userModel = new User();
    }

    // ... các method cũ (index, create, store, update)

    // Xử lý xoá user
    public function delete() {
        if (isset($_GET['id'])) {
            $id = (int)$_GET['id'];
            
            $result = $this->userModel->delete($id);

            if ($result) {
                header("Location: /");
                exit;
            } else {
                echo "Lỗi khi xoá dữ liệu!";
            }
        }
    }
}
